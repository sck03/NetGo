cmake_minimum_required(VERSION 3.25)
project(NetGoInstaller LANGUAGES CXX RC)
add_executable(netgo-maintenance WIN32 setup.cpp ../app.rc)
target_compile_features(netgo-maintenance PRIVATE cxx_std_17)
target_link_libraries(netgo-maintenance PRIVATE setupapi shell32 ole32 advapi32 user32)
if(MINGW)
  target_compile_options(netgo-maintenance PRIVATE -Wall -Wextra -Wpedantic -ffunction-sections -fdata-sections)
  target_link_options(netgo-maintenance PRIVATE -municode -static -static-libgcc -static-libstdc++ -Wl,--gc-sections -s)
endif()
if(DEFINED NETGO_PAYLOAD)
  file(TO_CMAKE_PATH "${NETGO_PAYLOAD}" payload)
  file(WRITE "${CMAKE_CURRENT_BINARY_DIR}/payload.rc" "#include <windows.h>\n201 RCDATA \"${payload}\"\n")
  add_executable(netgo-setup WIN32 setup.cpp ../app.rc "${CMAKE_CURRENT_BINARY_DIR}/payload.rc")
  target_compile_features(netgo-setup PRIVATE cxx_std_17)
  if(NETGO_DEFAULT_NETWORK)
    target_compile_definitions(netgo-setup PRIVATE NETGO_DEFAULT_NETWORK=1)
  endif()
  target_link_libraries(netgo-setup PRIVATE setupapi shell32 ole32 advapi32 user32)
  if(MINGW)
    target_compile_options(netgo-setup PRIVATE -Wall -Wextra -Wpedantic -ffunction-sections -fdata-sections)
    target_link_options(netgo-setup PRIVATE -municode -static -static-libgcc -static-libstdc++ -Wl,--gc-sections -s)
  endif()
endif()
