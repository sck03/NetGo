cmake_minimum_required(VERSION 3.25)
project(NetGoWindowsCapture LANGUAGES CXX)
if(NOT WIN32)
  message(FATAL_ERROR "This adapter requires the Windows desktop APIs")
endif()
add_library(netgo-capture STATIC src/desktop_capture.cpp)
target_include_directories(netgo-capture PUBLIC include)
target_compile_features(netgo-capture PUBLIC cxx_std_17)
target_compile_definitions(netgo-capture PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX UNICODE _UNICODE)
target_link_libraries(netgo-capture PRIVATE d3d11 dxgi user32)
if(MINGW)
  target_compile_options(netgo-capture PRIVATE -Wall -Wextra -Wpedantic -ffunction-sections -fdata-sections)
endif()

# Explicit opt-in, interactive desktop required. Do not register a screen capture
# as an unattended CTest; the probe writes metadata only, never screenshots.
option(NETGO_CAPTURE_PROBE "Build the interactive GPU capture probe" OFF)
if(NETGO_CAPTURE_PROBE)
  add_executable(netgo-capture-probe tests/capture_probe.cpp)
  target_link_libraries(netgo-capture-probe PRIVATE netgo-capture)
  if(MINGW)
    target_link_options(netgo-capture-probe PRIVATE -static -static-libgcc -static-libstdc++ -Wl,--gc-sections "$<$<CONFIG:Release>:-s>")
  endif()
endif()
