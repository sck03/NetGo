cmake_minimum_required(VERSION 3.25)
project(NetGoNative VERSION 0.3.0 LANGUAGES CXX RC)
include(../../scripts/cmake/NetGoVersion.cmake)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
set(NETGO_OPENSSL_COMPONENTS SSL Crypto)
include(../../scripts/cmake/NetGoOpenSSL.cmake)
find_path(JSON_INCLUDE nlohmann/json.hpp REQUIRED)

# The desktop only uses the OS, public profiles and local IPC. TLS, device
# signatures, connection state and the VPN process adapter belong to the agent.
add_library(netgo-platform STATIC src/platform.cpp src/profile.cpp src/ipc.cpp)
target_include_directories(netgo-platform PUBLIC include ${JSON_INCLUDE})
target_compile_definitions(netgo-platform PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(netgo-platform PUBLIC ws2_32 crypt32 advapi32 bcrypt)
add_library(netgo-connection STATIC src/transport.cpp src/control_protocol.cpp src/controller.cpp src/crypto.cpp src/engine.cpp src/engine_rpc.cpp)
target_link_libraries(netgo-connection PUBLIC netgo-platform PRIVATE OpenSSL::SSL OpenSSL::Crypto iphlpapi)
if(MINGW)
  target_compile_options(netgo-platform PUBLIC -Wall -Wextra -Wpedantic -ffunction-sections -fdata-sections)
  target_link_options(netgo-platform PUBLIC -static -static-libgcc -static-libstdc++ -Wl,--gc-sections)
endif()

add_subdirectory(remote)
add_subdirectory(../../packages/remote-core remote-core)
add_subdirectory(../../packages/remote-ticket remote-ticket)
add_subdirectory(../../packages/remote-transport remote-transport)
add_subdirectory(../../packages/remote-client remote-client)
add_executable(netgo WIN32 src/main.cpp src/agent.cpp src/ui.cpp src/ui_status.cpp src/profile_dialog.cpp remote/src/remote_runtime.cpp remote/src/remote_window.cpp app.rc "${NETGO_VERSION_RESOURCE}")
set_target_properties(netgo PROPERTIES OUTPUT_NAME NetGo)
target_include_directories(netgo PRIVATE remote/include)
target_link_libraries(netgo PRIVATE netgo-connection netgo-capture netgo-video netgo-audio netgo-remote-client shell32 comctl32 comdlg32 uxtheme gdi32 imm32)
if(MINGW)
  target_link_options(netgo PRIVATE -municode "$<$<CONFIG:Release>:-s>")
endif()

include(CheckIPOSupported)
check_ipo_supported(RESULT netgo_ipo_supported OUTPUT netgo_ipo_error LANGUAGES CXX)
if(netgo_ipo_supported)
  set_property(TARGET netgo-platform netgo-connection netgo PROPERTY INTERPROCEDURAL_OPTIMIZATION_RELEASE TRUE)
else()
  message(STATUS "Native link-time optimization unavailable: ${netgo_ipo_error}")
endif()

if(BUILD_TESTING)
  add_executable(native-tests tests/core_tests.cpp)
  target_link_libraries(native-tests PRIVATE netgo-connection OpenSSL::Crypto)
  add_test(NAME native-core COMMAND native-tests)
  add_test(NAME native-ipc COMMAND native-tests --ipc $<TARGET_FILE:netgo>)
  add_test(NAME native-profile-contract COMMAND native-tests --profile-contract ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/contracts/profiles.json)
endif()
