cmake_minimum_required(VERSION 3.25)
project(NetGoNative VERSION 0.2.3 LANGUAGES CXX RC)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(OPENSSL_USE_STATIC_LIBS TRUE)
find_package(OpenSSL REQUIRED)
find_path(JSON_INCLUDE nlohmann/json.hpp REQUIRED)
add_library(netgo-native src/platform.cpp src/transport.cpp src/controller.cpp)
target_include_directories(netgo-native PUBLIC include ${JSON_INCLUDE})
target_compile_definitions(netgo-native PUBLIC UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX _WIN32_WINNT=0x0A00)
target_link_libraries(netgo-native PUBLIC OpenSSL::SSL OpenSSL::Crypto ws2_32 crypt32 advapi32 iphlpapi bcrypt gdi32)
if(MINGW)
  target_compile_options(netgo-native PUBLIC -Wall -Wextra -Wpedantic)
  target_link_options(netgo-native PUBLIC -static -static-libgcc -static-libstdc++)
endif()
add_executable(netgo-agent WIN32 src/agent.cpp app.rc)
add_executable(netgo-desktop WIN32 src/ui.cpp app.rc)
target_link_libraries(netgo-agent PRIVATE netgo-native shell32)
target_link_libraries(netgo-desktop PRIVATE netgo-native comctl32 comdlg32 shell32 uxtheme)
if(MINGW)
  target_link_options(netgo-agent PRIVATE -municode)
  target_link_options(netgo-desktop PRIVATE -municode)
endif()
enable_testing()
add_executable(native-tests tests/core_tests.cpp)
target_link_libraries(native-tests PRIVATE netgo-native)
add_test(NAME native-core COMMAND native-tests)
add_test(NAME native-ipc COMMAND native-tests --ipc $<TARGET_FILE:netgo-agent>)
add_test(NAME native-profile-contract COMMAND native-tests --profile-contract ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/contracts/profiles.json)
