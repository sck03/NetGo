cmake_minimum_required(VERSION 3.25)
project(NetGoLinux VERSION 0.3.0 LANGUAGES C CXX)
include(../../scripts/cmake/NetGoVersion.cmake)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(CTest)
find_package(PkgConfig REQUIRED)
pkg_check_modules(GTK REQUIRED IMPORTED_TARGET gtk4)
pkg_check_modules(AV REQUIRED IMPORTED_TARGET libavcodec libavutil libswscale)
pkg_check_modules(CURL REQUIRED IMPORTED_TARGET libcurl)
pkg_check_modules(HOST REQUIRED IMPORTED_TARGET gstreamer-app-1.0 gio-unix-2.0 x11 xtst)
find_path(JSON_INCLUDE nlohmann/json.hpp REQUIRED)
add_subdirectory(../../packages/remote-core remote-core)
add_subdirectory(../../packages/remote-ticket remote-ticket)
add_subdirectory(../../packages/remote-transport remote-transport)
add_subdirectory(../../packages/remote-client remote-client)
add_executable(netgo-desktop src/main.cpp src/decoder.cpp src/ipc.cpp src/host.cpp src/startup.cpp src/service.cpp)
target_include_directories(netgo-desktop PRIVATE include ${JSON_INCLUDE})
target_link_libraries(netgo-desktop PRIVATE netgo-remote-client PkgConfig::GTK PkgConfig::AV PkgConfig::CURL PkgConfig::HOST)
target_compile_options(netgo-desktop PRIVATE -Wall -Wextra -Wpedantic -Werror)
install(TARGETS netgo-desktop RUNTIME DESTINATION lib/netgo)
install(FILES netgo.desktop DESTINATION share/applications)
