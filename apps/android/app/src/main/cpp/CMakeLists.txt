cmake_minimum_required(VERSION 3.25)
project(NetGoAndroidRemote LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
get_filename_component(NETGO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../../../../.." ABSOLUTE)
set(JSON_INCLUDE "${NETGO_ROOT}/.tools/native-json" CACHE PATH "Pinned JSON headers" FORCE)
set(OPENSSL_ROOT_DIR "${NETGO_ROOT}/.tools/openssl-android-arm64")
set(OPENSSL_INCLUDE_DIR "${OPENSSL_ROOT_DIR}/include" CACHE PATH "" FORCE)
set(OPENSSL_CRYPTO_LIBRARY "${OPENSSL_ROOT_DIR}/lib/libcrypto.a" CACHE FILEPATH "" FORCE)
set(OPENSSL_SSL_LIBRARY "${OPENSSL_ROOT_DIR}/lib/libssl.a" CACHE FILEPATH "" FORCE)
set(OPENSSL_USE_STATIC_LIBS TRUE)
add_subdirectory("${NETGO_ROOT}/packages/remote-core" remote-core)
add_subdirectory("${NETGO_ROOT}/packages/remote-ticket" remote-ticket)
add_subdirectory("${NETGO_ROOT}/packages/remote-transport" remote-transport)
add_subdirectory("${NETGO_ROOT}/packages/remote-client" remote-client)
add_library(netgo-remote-jni SHARED bridge.cpp)
target_include_directories(netgo-remote-jni PRIVATE ${JSON_INCLUDE})
target_link_libraries(netgo-remote-jni PRIVATE netgo-remote-client)
target_compile_options(netgo-remote-jni PRIVATE -Wall -Wextra -Wpedantic -Werror)
