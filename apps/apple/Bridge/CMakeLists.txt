cmake_minimum_required(VERSION 3.25)
project(NetGoAppleBridge LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(BUILD_TESTING OFF CACHE BOOL "" FORCE)
get_filename_component(NETGO_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/../../.." ABSOLUTE)
set(JSON_INCLUDE "${NETGO_ROOT}/.tools/native-json" CACHE PATH "JSON headers")
add_subdirectory("${NETGO_ROOT}/packages/remote-core" remote-core)
add_subdirectory("${NETGO_ROOT}/packages/remote-ticket" remote-ticket)
add_subdirectory("${NETGO_ROOT}/packages/remote-transport" remote-transport)
add_subdirectory("${NETGO_ROOT}/packages/remote-client" remote-client)
add_library(netgo-apple STATIC bridge.cpp)
target_include_directories(netgo-apple PUBLIC .)
target_link_libraries(netgo-apple PUBLIC netgo-remote-client)
