cmake_minimum_required(VERSION 3.25)
project(NetGoRemoteClient LANGUAGES CXX)
find_path(JSON_INCLUDE nlohmann/json.hpp REQUIRED)
add_library(netgo-remote-client STATIC src/client.cpp src/input.cpp src/host.cpp src/files.cpp src/exchange.cpp)
target_compile_features(netgo-remote-client PUBLIC cxx_std_17)
target_include_directories(netgo-remote-client PUBLIC include ${JSON_INCLUDE})
target_link_libraries(netgo-remote-client PUBLIC netgo-remote-core netgo-remote-ticket netgo-remote-transport)
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(netgo-remote-client PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(BUILD_TESTING)
 add_executable(remote-files-tests tests/files_tests.cpp)
 target_link_libraries(remote-files-tests PRIVATE netgo-remote-client)
 if(MINGW)
  target_link_options(remote-files-tests PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
 add_test(NAME remote-files COMMAND remote-files-tests)
 add_executable(remote-watchdog-tests tests/watchdog_tests.cpp)
 target_link_libraries(remote-watchdog-tests PRIVATE netgo-remote-client)
 if(MINGW)
  target_link_options(remote-watchdog-tests PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
 add_test(NAME remote-watchdog COMMAND remote-watchdog-tests)
 add_executable(remote-input-tests tests/input_tests.cpp)
 target_link_libraries(remote-input-tests PRIVATE netgo-remote-client)
 if(MINGW)
  target_link_options(remote-input-tests PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
 add_test(NAME remote-input COMMAND remote-input-tests)
 add_executable(remote-client-tests tests/client_tests.cpp)
 target_include_directories(remote-client-tests PRIVATE ${JSON_INCLUDE})
 target_link_libraries(remote-client-tests PRIVATE netgo-remote-client)
 if(MINGW)
  target_link_options(remote-client-tests PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
 add_test(NAME remote-client COMMAND remote-client-tests)
endif()
