cmake_minimum_required(VERSION 3.25)
project(NetGoRemoteCore VERSION 0.1.0 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
add_library(netgo-remote-core STATIC src/session.cpp)
target_include_directories(netgo-remote-core PUBLIC include)
set_target_properties(netgo-remote-core PROPERTIES POSITION_INDEPENDENT_CODE ON)
if(MSVC)
  target_compile_options(netgo-remote-core PRIVATE /W4 /WX)
else()
  target_compile_options(netgo-remote-core PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  add_executable(remote-session-tests tests/session_tests.cpp)
  target_link_libraries(remote-session-tests PRIVATE netgo-remote-core Threads::Threads)
  add_executable(remote-c-abi-test tests/c_abi_test.c)
  target_link_libraries(remote-c-abi-test PRIVATE netgo-remote-core)
  add_test(NAME remote-session COMMAND remote-session-tests)
  add_test(NAME remote-c-abi COMMAND remote-c-abi-test)
endif()
