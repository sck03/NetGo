cmake_minimum_required(VERSION 3.25)
project(NetGoRemoteTransport LANGUAGES CXX)
set(NETGO_OPENSSL_COMPONENTS SSL Crypto)
include(../../scripts/cmake/NetGoOpenSSL.cmake)
add_library(netgo-remote-transport STATIC src/transport.cpp)
target_compile_features(netgo-remote-transport PUBLIC cxx_std_17)
target_include_directories(netgo-remote-transport PUBLIC include)
target_link_libraries(netgo-remote-transport PUBLIC OpenSSL::SSL OpenSSL::Crypto)
if(WIN32)
 target_link_libraries(netgo-remote-transport PUBLIC ws2_32 crypt32 bcrypt)
endif()
if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
 target_compile_options(netgo-remote-transport PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
if(NETGO_TRANSPORT_TESTS)
 enable_testing()
 add_executable(netgo-remote-transport-test tests/transport_tests.cpp)
 target_link_libraries(netgo-remote-transport-test PRIVATE netgo-remote-transport)
 if(MINGW)
  target_link_options(netgo-remote-transport-test PRIVATE -static -static-libgcc -static-libstdc++)
 endif()
 add_test(NAME remote-transport COMMAND netgo-remote-transport-test)
endif()
