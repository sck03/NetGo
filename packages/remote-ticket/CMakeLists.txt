cmake_minimum_required(VERSION 3.25)
project(NetGoRemoteTicket LANGUAGES C CXX)
set(NETGO_OPENSSL_COMPONENTS Crypto)
include(../../scripts/cmake/NetGoOpenSSL.cmake)
add_library(netgo-remote-ticket STATIC src/ticket.cpp)
target_include_directories(netgo-remote-ticket PUBLIC include ../remote-core/include)
target_compile_features(netgo-remote-ticket PUBLIC cxx_std_17)
set_target_properties(netgo-remote-ticket PROPERTIES POSITION_INDEPENDENT_CODE ON CXX_EXTENSIONS OFF)
target_link_libraries(netgo-remote-ticket PRIVATE OpenSSL::Crypto)
if(MINGW)
  target_link_options(netgo-remote-ticket PUBLIC -static -static-libgcc -static-libstdc++)
endif()
if(MSVC)
  target_compile_options(netgo-remote-ticket PRIVATE /W4 /WX)
else()
  target_compile_options(netgo-remote-ticket PRIVATE -Wall -Wextra -Wpedantic -Werror)
endif()
include(CTest)
if(BUILD_TESTING)
  find_package(Threads REQUIRED)
  if(NOT TARGET netgo-remote-core)
    add_subdirectory(../remote-core ${CMAKE_CURRENT_BINARY_DIR}/remote-core)
  endif()
  add_executable(remote-ticket-tests tests/ticket_tests.cpp)
  target_link_libraries(remote-ticket-tests PRIVATE netgo-remote-ticket netgo-remote-core Threads::Threads OpenSSL::Crypto)
  add_executable(remote-ticket-c-abi tests/c_abi_test.c)
  target_link_libraries(remote-ticket-c-abi PRIVATE netgo-remote-ticket)
  set_target_properties(remote-ticket-c-abi PROPERTIES C_STANDARD 11 C_STANDARD_REQUIRED ON C_EXTENSIONS OFF)
  foreach(test_target remote-ticket-tests remote-ticket-c-abi)
    if(MSVC)
      target_compile_options(${test_target} PRIVATE /W4 /WX)
    else()
      target_compile_options(${test_target} PRIVATE -Wall -Wextra -Wpedantic -Werror)
    endif()
  endforeach()
  add_test(NAME remote-ticket COMMAND remote-ticket-tests ${CMAKE_CURRENT_SOURCE_DIR}/../../tests/contracts/remote/tickets.txt ${OPENSSL_VERSION})
  add_test(NAME remote-ticket-c-abi COMMAND remote-ticket-c-abi)
endif()
